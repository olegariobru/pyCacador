SKIP_DIRS = {
    'node_modules', '.git', '__pycache__', '.venv', 'venv',
    'windows', 'program files', 'program files (x86)',
    'appdata', '$recycle.bin', 'system volume information',
    'temp', 'tmp'
}

def format_size(b):
    if b >= 1073741824: return "{:.1f} GB".format(b / 1073741824.0)
    if b >= 1048576:    return "{:.1f} MB".format(b / 1048576.0)
    if b >= 1024:       return "{:.0f} KB".format(b / 1024.0)
    return "{} B".format(b)

def should_skip(path):
    parts = path.lower().replace('\\', '/').split('/')
    for part in parts:
        if part in SKIP_DIRS:
            return True
    return False

def sha256_file(path):
    h = hashlib.sha256()
    try:
        with open(path, 'rb') as f:
            while True:
                chunk = f.read(65536)
                if not chunk:
                    break
                h.update(chunk)
        return h.hexdigest()
    except Exception:
        return None

def find_duplicates(folder, min_size, extension, progress_cb, cancel_flag):
    """
    Retorna (grupos, erros)
    grupos = lista de dicts {hash, size, paths, recoverable}
    """
    # Passo 1: listar arquivos
    progress_cb('status', 'Listando arquivos...')
    all_files = []
    for root, dirs, files in os.walk(folder):
        if cancel_flag[0]: return [], []
        if should_skip(root): continue
        for fname in files:
            if fname.startswith('.'): continue
            if extension and not fname.lower().endswith(extension.lower()): continue
            fpath = os.path.join(root, fname)
            try:
                size = os.path.getsize(fpath)
                if size >= min_size:
                    all_files.append((fpath, size))
            except Exception:
                pass

    progress_cb('total_files', len(all_files))

    # Passo 2: agrupar por tamanho
    progress_cb('status', 'Agrupando por tamanho...')
    size_map = {}
    for fpath, size in all_files:
        size_map.setdefault(size, []).append(fpath)

    candidates = []
    for size, paths in size_map.items():
        if len(paths) > 1:
            for p in paths:
                candidates.append((p, size))

    if not candidates:
        return [], []

    # Passo 3: calcular hashes
    progress_cb('status', 'Calculando hashes...')
    total = len(candidates)
    hash_map = {}
    errors = []
    start = time.time()

    for i, (fpath, size) in enumerate(candidates):
        if cancel_flag[0]: return [], []

        pct = int((i + 1) / float(total) * 100)
        elapsed = time.time() - start
        if pct > 0:
            rem = int(elapsed / pct * (100 - pct))
        else:
            rem = 0
        rem_str = "{:02d}:{:02d}".format(rem // 60, rem % 60)
        progress_cb('hash_progress', (i + 1, total, pct, rem_str))

        h = sha256_file(fpath)
        if h:
            hash_map.setdefault(h, []).append((fpath, size))
        else:
            errors.append(fpath)

    # Passo 4: montar grupos
    grupos = []
    for h, items in hash_map.items():
        if len(items) < 2: continue
        size = items[0][1]
        paths = [p for p, _ in items]
        recoverable = size * (len(paths) - 1)
        grupos.append({
            'hash': h,
            'size': size,
            'paths': paths,
            'recoverable': recoverable,
        })

    # Ordenar por espaco recuperavel decrescente
    grupos.sort(key=lambda g: g['recoverable'], reverse=True)
    return grupos, errors

    # ── Selecao de pasta ─────────────────

    def _pick_folder(self):
        folder = filedialog.askdirectory(title='Selecione a pasta para escanear')
        if folder:
            self.var_path.set(folder)

    # ── Varredura ────────────────────────

    def _start_scan(self):
        folder = self.var_path.get().strip()
        if not folder:
            folder = filedialog.askdirectory(title='Selecione a pasta para escanear')
            if not folder: return
            self.var_path.set(folder)

        if not os.path.isdir(folder):
            messagebox.showerror('Erro', 'Pasta invalida: ' + folder)
            return

        try:
            min_size = int(self.var_minsize.get())
        except ValueError:
            min_size = 1024

        ext = self.var_ext.get().strip()
        if ext and not ext.startswith('.'):
            ext = '.' + ext

        self.grupos = []
        self.errors = []
        self.cancel_flag = [False]
        self._clear_table()
        self.btn_scan.config(state='disabled')
        self.btn_cancel.config(state='normal')
        self.btn_export.config(state='disabled')
        self.btn_delete.config(state='disabled')
        self.lbl_summary.config(text='')
        self.progressbar['value'] = 0

        self.scan_thread = threading.Thread(
            target=self._scan_worker,
            args=(folder, min_size, ext)
        )
        self.scan_thread.daemon = True
        self.scan_thread.start()

    def _scan_worker(self, folder, min_size, ext):
        def progress(kind, val):
            self.after(0, lambda: self._on_progress(kind, val))

        grupos, errors = find_duplicates(folder, min_size, ext, progress, self.cancel_flag)
        self.grupos = grupos
        self.errors = errors
        self.after(0, self._on_scan_done)

    def _on_progress(self, kind, val):
        if kind == 'status':
            self.lbl_status.config(text='  ' + val)
        elif kind == 'total_files':
            self.lbl_status.config(text='  {} arquivos encontrados'.format(val))
        elif kind == 'hash_progress':
            done, total, pct, rem = val
            self.lbl_status.config(
                text='  Calculando hashes... {}/{} ({}%) - restam ~{}'.format(done, total, pct, rem))
            self.progressbar['value'] = pct

    def _on_scan_done(self):
        self.btn_scan.config(state='normal')
        self.btn_cancel.config(state='disabled')

        if self.cancel_flag[0]:
            self.lbl_status.config(text='  Varredura cancelada.')
            self.progressbar['value'] = 0
            return

        self.progressbar['value'] = 100

        if not self.grupos:
            self.lbl_status.config(text='  Nenhuma duplicata encontrada.')
            messagebox.showinfo('Resultado', 'Nenhuma duplicata encontrada!')
            return

        self._populate_table()
        self.btn_export.config(state='normal')
        self.btn_delete.config(state='normal')

        # Resumo
        total_rec  = sum(g['recoverable'] for g in self.grupos)
        total_dup  = sum(len(g['paths']) for g in self.grupos)
        self.lbl_summary.config(
            text='  {} grupos  |  {} arquivos duplicados  |  {} recuperaveis'.format(
                len(self.grupos), total_dup, format_size(total_rec)))
        self.lbl_status.config(text='  Varredura concluida.')

        if self.errors:
            self.lbl_status.config(
                text='  Concluido com {} arquivo(s) inacessivel(is).'.format(len(self.errors)))

        # Modo silencioso: exportar automaticamente
        if self.var_silent.get():
            self._export(silent=True)

    def _cancel_scan(self):
        self.cancel_flag[0] = True
        self.lbl_status.config(text='  Cancelando...')

    # ── Tabela ───────────────────────────

    def _clear_table(self):
        for item in self.tree.get_children():
            self.tree.delete(item)

    def _populate_table(self):
        self._clear_table()
        row_idx = 0
        for gi, grupo in enumerate(self.grupos):
            rec_str  = format_size(grupo['recoverable'])
            size_str = format_size(grupo['size'])
            short_hash = grupo['hash'][:16] + '...'

            # Linha de cabecalho do grupo
            self.tree.insert('', 'end',
                iid='g{}'.format(gi),
                values=('Grupo {}'.format(gi + 1),
                        '{} copias  |  {} cada  |  {} recuperaveis'.format(
                            len(grupo['paths']), size_str, rec_str),
                        '', '', short_hash),
                tags=('header',))

            # Arquivos do grupo
            for pi, path in enumerate(grupo['paths']):
                tag = 'even' if row_idx % 2 == 0 else 'odd'
                self.tree.insert('', 'end',
                    iid='g{}p{}'.format(gi, pi),
                    values=('', path, size_str, rec_str if pi > 0 else '(original)', short_hash),
                    tags=(tag,))
                row_idx += 1

        pass  # Pode ser expandido futuramente

    # ── Duplo clique: abrir pasta ────────

    def _open_folder(self, event):
        sel = self.tree.selection()
        if not sel: return
        item = self.tree.item(sel[0])
        path = item['values'][1] if item['values'] else ''
        if os.path.isfile(str(path)):
            folder = os.path.dirname(str(path))
            try:
                os.startfile(folder)
            except Exception as e:
                messagebox.showerror('Erro', str(e))

    # ── Menu de contexto (botao direito) ─

    def _ctx_menu(self, event):
        sel = self.tree.identify_row(event.y)
        if not sel: return
        self.tree.selection_set(sel)
        item = self.tree.item(sel)
        path = str(item['values'][1]) if item['values'] else ''
        if not os.path.isfile(path): return

        menu = tk.Menu(self, tearoff=0, bg=self.PANEL, fg=self.TEXT,
                       activebackground=self.ACCENT, font=('Segoe UI', 9))
        menu.add_command(label='Abrir pasta',
                         command=lambda: os.startfile(os.path.dirname(path)))
        menu.add_command(label='Deletar este arquivo',
                         command=lambda: self._delete_one(path))
        menu.add_separator()
        menu.add_command(label='Copiar caminho',
                         command=lambda: self._copy_to_clipboard(path))
        menu.post(event.x_root, event.y_root)

    def _copy_to_clipboard(self, text):
        self.clipboard_clear()
        self.clipboard_append(text)

    # ── Delecao ──────────────────────────

    def _delete_one(self, path):
        if messagebox.askyesno('Confirmar', 'Deletar este arquivo?\n\n' + path):
            try:
                os.remove(path)
                messagebox.showinfo('OK', 'Arquivo deletado.')
                self._refresh_after_delete(path)
            except Exception as e:
                messagebox.showerror('Erro', str(e))

    def _delete_selected(self):
        sel = self.tree.selection()
        paths = []
        for iid in sel:
            item = self.tree.item(iid)
            path = str(item['values'][1]) if item['values'] else ''
            if os.path.isfile(path):
                paths.append(path)

        if not paths:
            messagebox.showwarning('Aviso', 'Selecione arquivos na lista para deletar.\n'
                                            'Use Ctrl+clique para selecionar varios.\n'
                                            'Linhas de cabecalho de grupo sao ignoradas.')
            return

        msg = 'Deletar {} arquivo(s) selecionado(s)?\n\n'.format(len(paths))
        msg += '\n'.join(paths[:10])
        if len(paths) > 10:
            msg += '\n... e mais {}'.format(len(paths) - 10)

        if not messagebox.askyesno('Confirmar delecao', msg):
            return

        deleted = 0
        failed  = []
        for path in paths:
            try:
                os.remove(path)
                self._refresh_after_delete(path)
                deleted += 1
            except Exception as e:
                failed.append('{} ({})'.format(path, e))

        msg = '{} arquivo(s) deletado(s).'.format(deleted)
        if failed:
            msg += '\n\nFalhou:\n' + '\n'.join(failed)
        messagebox.showinfo('Resultado', msg)

    def _refresh_after_delete(self, deleted_path):
        # Remove da tabela
        for iid in self.tree.get_children():
            item = self.tree.item(iid)
            if str(item['values'][1]) == deleted_path:
                self.tree.delete(iid)
                break
        # Remove dos grupos internos
        for grupo in self.grupos:
            if deleted_path in grupo['paths']:
                grupo['paths'].remove(deleted_path)

    # ── Exportar ─────────────────────────

    def _export(self, silent=False):
        if not self.grupos:
            if not silent:
                messagebox.showwarning('Aviso', 'Nenhum resultado para exportar.')
            return

        timestamp = datetime.now().strftime('%Y-%m-%d_%H-%M-%S')

        if silent:
            script_dir = os.path.dirname(os.path.abspath(sys.argv[0]))
            base = os.path.join(script_dir, 'duplicatas_' + timestamp)
        else:
            base = filedialog.asksaveasfilename(
                title='Salvar relatorio',
                defaultextension='.txt',
                initialfile='duplicatas_' + timestamp,
                filetypes=[('Texto', '*.txt'), ('CSV', '*.csv'), ('Todos', '*.*')]
            )
            if not base: return
            if base.endswith('.txt') or base.endswith('.csv'):
                base = base[:-4]

        txt_path = base + '.txt'
        csv_path = base + '.csv'
        err_path = base + '_erros.txt'

        # TXT
        lines = []
        lines.append('CACADOR DE DUPLICATAS - Relatorio')
        lines.append('Gerado em: ' + datetime.now().strftime('%d/%m/%Y %H:%M:%S'))
        lines.append('=' * 60)
        lines.append('')
        total_rec = 0
        total_dup = 0
        for gi, grupo in enumerate(self.grupos):
            total_rec += grupo['recoverable']
            total_dup += len(grupo['paths'])
            lines.append('Grupo {} -- {} copias -- {} recuperaveis'.format(
                gi + 1, len(grupo['paths']), format_size(grupo['recoverable'])))
            lines.append('SHA-256: ' + grupo['hash'])
            for path in grupo['paths']:
                lines.append('    ' + path)
            lines.append('')
        lines.append('=' * 60)
        lines.append('Grupos de duplicatas : {}'.format(len(self.grupos)))
        lines.append('Arquivos duplicados  : {}'.format(total_dup))
        lines.append('Espaco recuperavel   : {}'.format(format_size(total_rec)))

        try:
            with open(txt_path, 'w') as f:
                f.write('\n'.join(lines))
        except Exception as e:
            if not silent:
                messagebox.showerror('Erro', 'Nao foi possivel salvar TXT:\n' + str(e))

        # CSV
        try:
            with open(csv_path, 'w') as f:
                writer = csv.writer(f, delimiter=';')
                writer.writerow(['Grupo', 'Arquivo', 'Tamanho', 'Recuperavel', 'Hash'])
                for gi, grupo in enumerate(self.grupos):
                    for path in grupo['paths']:
                        writer.writerow([
                            gi + 1, path,
                            format_size(grupo['size']),
                            format_size(grupo['recoverable']),
                            grupo['hash']
                        ])
        except Exception as e:
            if not silent:
                messagebox.showerror('Erro', 'Nao foi possivel salvar CSV:\n' + str(e))

        # Log de erros
        if self.errors:
            try:
                with open(err_path, 'w') as f:
                    f.write('\n'.join(self.errors))
            except Exception:
                pass

        if not silent:
            msg = 'Relatorio salvo:\n\n  {}\n  {}'.format(txt_path, csv_path)
            if self.errors:
                msg += '\n  {} (erros)'.format(err_path)
            messagebox.showinfo('Exportado!', msg)
